async function hcAdminApi(action, payload = {}) { const base = "https://harshiyaan-collections-store.vercel.app"; const url = base + "/api/order-sync"; const response = await fetch(url, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ action, ...payload }) }); const data = await response.json().catch(() => ({})); if (!response.ok) { throw new Error( data.error || `Order sync failed: HTTP ${response.status}` ); } return data; } async function loadCentralOrders() { const statusElement = document.getElementById("hcCloudSyncStatus"); try { statusElement.textContent = "Connecting to central order storage…"; const base = "https://harshiyaan-collections-store.vercel.app"; const response = await fetch( base + "/api/order-sync?action=list", { cache: "no-store" } ); const data = await response.json().catch(() => ({})); if (!response.ok) { throw new Error( data.error || `HTTP ${response.status}` ); } if (!data.ok || !Array.isArray(data.orders)) { throw new Error("Unexpected response from central order storage"); } orders = data.orders; save("hc_orders", orders); statusElement.textContent = "✓ Central order storage connected"; document.getElementById("statOrders").textContent = orders.length; renderOrders(); } catch (error) { console.error("Central order sync failed:", error); statusElement.textContent = "⚠ Central sync unavailable; showing local orders. " + (error.message || "Please check the connection."); } }
